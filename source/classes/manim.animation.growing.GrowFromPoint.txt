class GrowFromPointExample(Scene):
 def construct(self):
 dot = Dot(3 * UR, color=GREEN)
 squares = [Square() for _ in range(4)]
 VGroup(*squares).set_x(0).arrange(buff=1)
 self.add(dot)
 self.play(GrowFromPoint(squares[0], ORIGIN))
